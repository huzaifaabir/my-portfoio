import { Field, Input, Textarea } from '../Form.jsx';

// Supported field types: text, textarea, range, select
// Select options: [{ value, label }]
// Set unique: true to prevent selecting the same option in multiple rows.

export default function ListEditor({
  items,
  onChange,
  fields,
  emptyItem,
  addLabel = 'Add item',
  disabled = false,
  itemLabel = 'Item',
}) {
  const update = (index, key, value) => {
    onChange(
      items.map((item, i) =>
        i === index ? { ...item, [key]: value } : item
      )
    );
  };

  const remove = (index) => {
    onChange(items.filter((_, i) => i !== index));
  };

  const add = () => {
    onChange([...items, { ...emptyItem }]);
  };

  const uniqueSelectField = fields.find(
    (field) => field.type === 'select' && field.unique
  );

  const canAddMore =
    !uniqueSelectField ||
    items.length < (uniqueSelectField.options || []).length;

  return (
    <div className="space-y-4">
      {items.length === 0 && (
        <p className="text-sm text-muted">Nothing added yet.</p>
      )}

      {items.map((item, index) => (
        <div
          key={index}
          className="rounded-xl border border-line bg-surface-2/60 p-4"
        >
          <div className="mb-4 flex items-center justify-between">
            <p className="text-sm font-semibold text-muted">
              {itemLabel} {index + 1}
            </p>

            {!disabled && (
              <button
                type="button"
                className="text-sm text-danger hover:underline"
                onClick={() => remove(index)}
              >
                Remove
              </button>
            )}
          </div>

          <div className="grid gap-4 sm:grid-cols-2">
            {fields.map((f) => {
              const id = `${f.key}-${index}`;

              if (f.type === 'select') {
                const selectedElsewhere = items
                  .filter((_, i) => i !== index)
                  .map((other) => other[f.key]);

                return (
                  <div key={f.key}>
                    <Field label={f.label} htmlFor={id}>
                      <select
                        id={id}
                        disabled={disabled}
                        value={item[f.key] ?? ''}
                        onChange={(e) =>
                          update(index, f.key, e.target.value)
                        }
                        className="input w-full"
                      >
                        <option value="" disabled>
                          {f.placeholder || 'Select an option'}
                        </option>

                        {(f.options || []).map((option) => (
                          <option
                            key={option.value}
                            value={option.value}
                            disabled={
                              f.unique &&
                              selectedElsewhere.includes(option.value)
                            }
                          >
                            {option.label}
                          </option>
                        ))}
                      </select>
                    </Field>
                  </div>
                );
              }

              return (
                <div
                  key={f.key}
                  className={f.type === 'textarea' ? 'sm:col-span-2' : ''}
                >
                  <Field label={f.label} htmlFor={id}>
                    {f.type === 'textarea' ? (
                      <Textarea
                        id={id}
                        rows={3}
                        disabled={disabled}
                        placeholder={f.placeholder}
                        maxLength={f.max}
                        value={item[f.key] ?? ''}
                        onChange={(e) =>
                          update(index, f.key, e.target.value)
                        }
                      />
                    ) : f.type === 'range' ? (
                      <div className="flex items-center gap-4">
                        <input
                          id={id}
                          type="range"
                          min="0"
                          max="100"
                          disabled={disabled}
                          value={item[f.key] ?? 50}
                          onChange={(e) =>
                            update(index, f.key, Number(e.target.value))
                          }
                          className="w-full accent-[var(--color-accent)]"
                        />

                        <span className="w-12 text-right text-sm tabular-nums">
                          {item[f.key] ?? 50}%
                        </span>
                      </div>
                    ) : (
                      <Input
                        id={id}
                        disabled={disabled}
                        placeholder={f.placeholder}
                        maxLength={f.max}
                        value={item[f.key] ?? ''}
                        onChange={(e) =>
                          update(index, f.key, e.target.value)
                        }
                      />
                    )}
                  </Field>
                </div>
              );
            })}
          </div>
        </div>
      ))}

      {!disabled && (
        <button
          type="button"
          className="btn-ghost btn-sm"
          onClick={add}
          disabled={!canAddMore}
        >
          + {addLabel}
        </button>
      )}
    </div>
  );
}

import { useEffect } from 'react';

// Accessible confirmation modal. Escape or the Cancel button closes it.
export default function ConfirmDialog({ open, title, message, confirmLabel = 'Confirm', danger = false, busy = false, onConfirm, onCancel }) {
  useEffect(() => {
    if (!open) return undefined;
    const onKey = (e) => e.key === 'Escape' && !busy && onCancel();
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, [open, busy, onCancel]);

  if (!open) return null;
  return (
    <div className="fixed inset-0 z-[90] flex items-center justify-center bg-black/70 p-4 backdrop-blur-sm" role="presentation" onClick={() => !busy && onCancel()}>
      <div role="dialog" aria-modal="true" aria-labelledby="confirm-title" className="card w-full max-w-md" onClick={(e) => e.stopPropagation()}>
        <h2 id="confirm-title" className="text-lg font-semibold">{title}</h2>
        {message && <p className="mt-2 text-sm text-muted">{message}</p>}
        <div className="mt-6 flex justify-end gap-3">
          <button className="btn-ghost btn-sm" onClick={onCancel} disabled={busy}>Cancel</button>
          <button className={`${danger ? 'btn-danger' : 'btn-primary'} btn-sm`} onClick={onConfirm} disabled={busy}>
            {busy ? 'Please wait...' : confirmLabel}
          </button>
        </div>
      </div>
    </div>
  );
}

// Form building blocks so every admin form looks and behaves the same way

export function Field({ label, htmlFor, error, help, required, children }) {
  return (
    <div>
      {label && (
        <label htmlFor={htmlFor} className="label">
          {label}
          {required && <span className="text-accent"> *</span>}
        </label>
      )}
      {children}
      {help && !error && <p className="help">{help}</p>}
      {error && <p role="alert" className="error-text">{error}</p>}
    </div>
  );
}

export function Input({ invalid, className = '', ...props }) {
  return (
    <input
      {...props}
      aria-invalid={invalid || undefined}
      className={`input ${invalid ? 'border-danger focus:border-danger' : ''} ${className}`}
    />
  );
}

export function Textarea({ invalid, className = '', rows = 4, ...props }) {
  return (
    <textarea
      rows={rows}
      {...props}
      aria-invalid={invalid || undefined}
      className={`input resize-y ${invalid ? 'border-danger' : ''} ${className}`}
    />
  );
}

export function Select({ className = '', children, ...props }) {
  return <select {...props} className={`input appearance-none pr-10 ${className}`}>{children}</select>;
}

// Checkbox with a label to the right
export function Toggle({ label, checked, onChange, disabled, id }) {
  return (
    <label htmlFor={id} className={`flex cursor-pointer items-center gap-3 text-sm ${disabled ? 'opacity-60' : ''}`}>
      <input
        id={id}
        type="checkbox"
        checked={checked}
        disabled={disabled}
        onChange={(e) => onChange(e.target.checked)}
        className="h-4 w-4 accent-[var(--color-accent)]"
      />
      {label}
    </label>
  );
}

import { useEffect, useRef, useState } from 'react';
import { staffApi } from '../lib/services.js';
import { useToast } from '../context/ToastContext.jsx';
import { errorMessage } from '../lib/format.js';

const MAX_BYTES = 5 * 1024 * 1024;
const TYPES = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
const CROP_SIZE = 320;

export default function ImageUploader({
  value,
  onChange,
  folder,
  label = 'Image',
  aspect = 'aspect-video',
  rounded = 'rounded-2xl',
  help,
}) {
  const inputRef = useRef(null);
  const dragRef = useRef(null);
  const sessionUploads = useRef(new Set());

  const toast = useToast();

  const [candidate, setCandidate] = useState(null);
  const [preview, setPreview] = useState('');
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState('');
  const [mode, setMode] = useState('square');
  const [zoom, setZoom] = useState(1);
  const [offset, setOffset] = useState({ x: 0, y: 0 });
  const [dragging, setDragging] = useState(false);

  useEffect(() => {
    return () => {
      if (candidate?.url) URL.revokeObjectURL(candidate.url);
    };
  }, [candidate?.url]);

  const isGif = candidate?.file?.type === 'image/gif';

  function resetCrop() {
    setZoom(1);
    setOffset({ x: 0, y: 0 });
  }

  function closeEditor() {
    setCandidate(null);
    setError('');
    resetCrop();
  }

  async function handleFile(e) {
    const file = e.target.files?.[0];
    e.target.value = '';

    if (!file) return;

    if (!TYPES.includes(file.type)) {
      setError('Please choose a JPG, PNG, WEBP or GIF image.');
      return;
    }

    if (file.size > MAX_BYTES) {
      setError('The image is larger than 5 MB. Please choose a smaller file.');
      return;
    }

    setError('');

    const url = URL.createObjectURL(file);
    const image = new Image();

    image.onload = () => {
      setCandidate({
        file,
        url,
        width: image.naturalWidth,
        height: image.naturalHeight,
      });

      setMode('square');
      resetCrop();
    };

    image.onerror = () => {
      URL.revokeObjectURL(url);
      setError('Unable to read this image. Please try another file.');
    };

    image.src = url;
  }

  function getCropBounds() {
    if (!candidate) return null;

    const { width, height } = candidate;
    const scale = (CROP_SIZE / Math.min(width, height)) * zoom;

    const cropSide = CROP_SIZE / scale;

    const centerX = width / 2 - offset.x / scale;
    const centerY = height / 2 - offset.y / scale;

    const sx = Math.max(0, Math.min(width - cropSide, centerX - cropSide / 2));
    const sy = Math.max(0, Math.min(height - cropSide, centerY - cropSide / 2));

    return { sx, sy, side: cropSide };
  }

  function startDrag(e) {
    if (mode !== 'square' || isGif) return;

    e.preventDefault();
    e.currentTarget.setPointerCapture(e.pointerId);

    dragRef.current = {
      x: e.clientX,
      y: e.clientY,
      offsetX: offset.x,
      offsetY: offset.y,
    };

    setDragging(true);
  }

  function moveDrag(e) {
    if (!dragRef.current || !candidate) return;

    const { width, height } = candidate;
    const scale = (CROP_SIZE / Math.min(width, height)) * zoom;

    const displayWidth = width * scale;
    const displayHeight = height * scale;

    const maxX = Math.max(0, (displayWidth - CROP_SIZE) / 2);
    const maxY = Math.max(0, (displayHeight - CROP_SIZE) / 2);

    const dx = e.clientX - dragRef.current.x;
    const dy = e.clientY - dragRef.current.y;

    setOffset({
      x: Math.max(-maxX, Math.min(maxX, dragRef.current.offsetX + dx)),
      y: Math.max(-maxY, Math.min(maxY, dragRef.current.offsetY + dy)),
    });
  }

  function stopDrag() {
    dragRef.current = null;
    setDragging(false);
  }

  async function createCroppedFile() {
    const bounds = getCropBounds();
    if (!bounds) throw new Error('Unable to crop this image.');

    const canvas = document.createElement('canvas');
    const side = Math.max(1, Math.floor(bounds.side));

    canvas.width = side;
    canvas.height = side;

    const image = new Image();
    image.src = candidate.url;

    await new Promise((resolve, reject) => {
      image.onload = resolve;
      image.onerror = () => reject(new Error('Unable to process the image.'));
    });

    const context = canvas.getContext('2d');

    if (!context) {
      throw new Error('Your browser does not support image cropping.');
    }

    context.drawImage(
      image,
      bounds.sx,
      bounds.sy,
      bounds.side,
      bounds.side,
      0,
      0,
      side,
      side
    );

    const mimeType =
      candidate.file.type === 'image/png'
        ? 'image/png'
        : candidate.file.type === 'image/webp'
          ? 'image/webp'
          : 'image/jpeg';

    const blob = await new Promise((resolve, reject) => {
      canvas.toBlob(
        (result) => {
          if (result) resolve(result);
          else reject(new Error('Could not create the cropped image.'));
        },
        mimeType,
        0.92
      );
    });

    const extension =
      mimeType === 'image/png'
        ? 'png'
        : mimeType === 'image/webp'
          ? 'webp'
          : 'jpg';

    return new File(
      [blob],
      `cropped-${Date.now()}.${extension}`,
      { type: mimeType }
    );
  }

  async function confirmUpload() {
    if (!candidate || uploading) return;

    setUploading(true);
    setError('');

    try {
      // Full Size sends the original file.
      // GIF is always uploaded as Full Size to preserve animation.
      const file =
        mode === 'square' && !isGif
          ? await createCroppedFile()
          : candidate.file;

      if (file.size > MAX_BYTES) {
        throw new Error('The processed image is larger than 5 MB.');
      }

      const uploaded = await staffApi.uploadImage(file, folder);

      sessionUploads.current.add(uploaded.path);
      onChange(uploaded);

      setPreview('');
      setCandidate(null);
      resetCrop();

      toast.success('Image uploaded successfully.');
    } catch (err) {
      setError(errorMessage(err));
    } finally {
      setUploading(false);
    }
  }

  async function remove() {
    const path = value?.path;

    if (path && sessionUploads.current.has(path)) {
      sessionUploads.current.delete(path);
      staffApi.deleteUpload(path).catch(() => {});
    }

    setPreview('');
    setError('');
    onChange({ url: '', path: '' });
  }

  function selectMode(nextMode) {
    setMode(nextMode);
    resetCrop();
  }

  const shown = preview || value?.url || '';

  const imageScale = candidate
    ? (CROP_SIZE / Math.min(candidate.width, candidate.height)) * zoom
    : 1;

  const imageWidth = candidate ? candidate.width * imageScale : CROP_SIZE;
  const imageHeight = candidate ? candidate.height * imageScale : CROP_SIZE;

  const imageLeft = (CROP_SIZE - imageWidth) / 2 + offset.x;
  const imageTop = (CROP_SIZE - imageHeight) / 2 + offset.y;

  return (
    <div className="space-y-3">
      {label && <span className="label">{label}</span>}

      <div
        className={`relative overflow-hidden border border-dashed border-line bg-surface-2 ${rounded} ${aspect} ${
          shown ? '' : 'flex items-center justify-center'
        }`}
      >
        {shown ? (
          <img
            src={shown}
            alt="Image preview"
            className="h-full w-full object-cover"
          />
        ) : (
          <div className="px-6 text-center text-sm text-muted">
            <p className="text-2xl">ðŸ–¼ï¸</p>
            <p className="mt-2">No image selected</p>
          </div>
        )}

        {uploading && (
          <div className="absolute inset-0 flex items-center justify-center bg-bg/70 backdrop-blur-sm">
            <p className="text-sm font-medium">Uploading...</p>
          </div>
        )}
      </div>

      <div className="flex flex-wrap gap-2">
        <input
          ref={inputRef}
          type="file"
          accept={TYPES.join(',')}
          className="sr-only"
          onChange={handleFile}
          aria-label={`Choose ${label}`}
        />

        <button
          type="button"
          className="btn-ghost btn-sm"
          onClick={() => inputRef.current?.click()}
          disabled={uploading}
        >
          {shown ? 'Change image' : 'Choose image'}
        </button>

        {shown && !uploading && (
          <button
            type="button"
            className="btn-ghost btn-sm text-danger"
            onClick={remove}
          >
            Remove
          </button>
        )}
      </div>

      {help && !error && <p className="help">{help}</p>}
      {error && <p role="alert" className="error-text">{error}</p>}

      {/* Image editing modal */}
      {candidate && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center overflow-y-auto bg-black/80 p-4">
          <div className="my-auto w-full max-w-lg rounded-2xl border border-line bg-surface p-5 shadow-2xl">
            <div className="mb-5 flex items-center justify-between gap-3">
              <div>
                <h3 className="text-lg font-semibold">Prepare your image</h3>
                <p className="mt-1 text-xs text-muted">
                  Choose how you want to upload this image.
                </p>
              </div>

              <button

  const ringRef = useRef(null);

  const labelRef = useRef(null);

  useEffect(() => {
    const finePointer = window.matchMedia('(pointer: fine)');
    const reducedMotion = window.matchMedia(
      '(prefers-reduced-motion: reduce)'
    );

    if (!finePointer.matches || reducedMotion.matches) return;

    const cursor = cursorRef.current;
    const dot = dotRef.current;
    const ring = ringRef.current;
    const aura = auraRef.current;
    const label = labelRef.current;

    if (!cursor || !dot || !ring || !aura || !label) return;

    let targetX = -100;
    let targetY = -100;

    let ringX = targetX;
    let ringY = targetY;

    let auraX = targetX;
    let auraY = targetY;

    let velocityX = 0;
    let velocityY = 0;

    let previousX = targetX;
    let previousY = targetY;

    let frameId = null;
    let active = false;

    const animate = () => {
      const dx = targetX - ringX;
      const dy = targetY - ringY;

      ringX += dx * 0.19;
      ringY += dy * 0.19;

      auraX += (targetX - auraX) * 0.09;
      auraY += (targetY - auraY) * 0.09;

      velocityX = targetX - previousX;
      velocityY = targetY - previousY;

      previousX = targetX;
      previousY = targetY;

      const speed = Math.min(
        Math.hypot(velocityX, velocityY) / 35,
        1
      );

      const angle = Math.atan2(velocityY, velocityX) * (180 / Math.PI);

      dot.style.left = `${targetX}px`;
      dot.style.top = `${targetY}px`;

      ring.style.left = `${ringX}px`;
      ring.style.top = `${ringY}px`;

      aura.style.top = `${auraY}px`;

import { useEffect, useRef, useState } from 'react';

export default function Reveal({
  children,
  className = '',
  delay = 0,
}) {
  const ref = useRef(null);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    const element = ref.current;

    if (!element || typeof IntersectionObserver === 'undefined') {
      setVisible(true);
      return undefined;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        setVisible(entry.isIntersecting);
      },
      {
        threshold: 0.12,
        rootMargin: '0px 0px -30px 0px',
      }
    );

    observer.observe(element);

    return () => {
      observer.disconnect();
    };
  }, []);

  return (
    <div
      ref={ref}
      className={`reveal ${visible ? 'is-visible' : ''} ${className}`}
      style={{
        transitionDelay: visible ? `${delay}ms` : '0ms',
      }}
    >
      {children}
    </div>
  );
}

// Friendly text for the Firebase error codes a user can actually hit
function loginErrorMessage(err) {
  switch (err?.code) {
    case 'auth/invalid-credential':
    case 'auth/wrong-password':
    case 'auth/user-not-found':
    case 'auth/invalid-email':
      return 'Email or password is incorrect.';
    case 'auth/too-many-requests':
      return 'Too many attempts. Please wait a few minutes and try again.';
    case 'auth/network-request-failed':
      return 'Network error. Check your internet connection.';
    default:
      return err?.message || 'Login failed. Please try again.';
  }
}

export function AuthProvider({ children }) {
  const [firebaseUser, setFirebaseUser] = useState(null);
  const [me, setMe] = useState(null); // staff profile + permissions from the API
  const [meError, setMeError] = useState('');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!auth) {
      setLoading(false);
      return undefined;
    }
    return onAuthStateChanged(auth, async (user) => {
      setFirebaseUser(user);
      if (user) {
        try {
          setMe(await staffApi.me());
          setMeError('');
        } catch (err) {
          setMe(null);
          setMeError(err instanceof ApiError ? err.message : 'Could not load your account.');
        }
      } else {
        setMe(null);
        setMeError('');
      }
      setLoading(false);
    });
  }, []);

  const login = useCallback(async (email, password) => {
    if (!auth) throw new Error('Firebase is not configured. Add the VITE_FIREBASE_* values to client/.env.');
    try {
      await signInWithEmailAndPassword(auth, email.trim(), password);
    } catch (err) {
      throw new Error(loginErrorMessage(err));
    }
    try {
      const profile = await staffApi.me();
      setMe(profile);
      setMeError('');
      return profile;
    } catch (err) {
      // Signed in to Firebase, but this email is not a staff member in our database
      await signOut(auth);
      throw new Error(err instanceof ApiError && err.status === 403
        ? 'This account does not have admin access.'
        : err.message || 'Could not verify your account.');
    }
  }, []);

  const logout = useCallback(() => (auth ? signOut(auth) : Promise.resolve()), []);

  // Re-read the staff profile after the user changes their own details
  const refreshMe = useCallback(async () => {
    setMe(await staffApi.me());
  }, []);

  const resetPassword = useCallback((email) => {
    if (!auth) throw new Error('Firebase is not configured yet.');
    return sendPasswordResetEmail(auth, email.trim());
  }, []);

  const value = useMemo(() => ({
    firebaseUser,
    me,
    meError,
    loading,
    isStaff: Boolean(me),
    can: (permission) => Boolean(me?.permissions?.[permission]),
    login,
    logout,
    resetPassword,
    refreshMe,
  }), [firebaseUser, me, meError, loading, login, logout, resetPassword, refreshMe]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

import { useCallback, useEffect, useState } from 'react';

// Loads data with loading / error / retry handling.
// Usage: const { data, loading, error, reload } = useAsync(() => api.call(), [dep])
export function useAsync(fn, deps = []) {
  const [state, setState] = useState({ data: null, loading: true, error: '' });
  const [version, setVersion] = useState(0);

  useEffect(() => {
    let alive = true;
    setState((s) => ({ ...s, loading: true, error: '' }));
    fn()
      .then((data) => alive && setState({ data, loading: false, error: '' }))
      .catch((err) => alive && setState({ data: null, loading: false, error: err.message || 'Could not load data.' }));
    return () => {
      alive = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, version]);

  const reload = useCallback(() => setVersion((v) => v + 1), []);
  const setData = useCallback((updater) => setState((s) => ({
    ...s,
    data: typeof updater === 'function' ? updater(s.data) : updater,
  })), []);

  return { ...state, reload, setData };
}

import { auth } from '../firebase.js';

// Empty in local dev (Vite proxy). Set VITE_API_URL to your deployed API in production.
const BASE = (import.meta.env.VITE_API_URL || '').replace(/\/$/, '');

export class ApiError extends Error {
  constructor(message, status = 0) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
  }
}

// One request helper for the whole app.
// auth: true attaches the Firebase ID token. formData: send a multipart upload.
export async function request(path, { method = 'GET', body, formData, auth: needsAuth = false } = {}) {
  const headers = {};
  if (needsAuth) {
    const user = auth?.currentUser;
    if (!user) throw new ApiError('Please log in to continue.', 401);
    headers.Authorization = `Bearer ${await user.getIdToken()}`;
  }

  let payload;
  if (formData) {
    payload = formData; // browser sets the multipart boundary itself
  } else if (body !== undefined) {
    headers['Content-Type'] = 'application/json';
    payload = JSON.stringify(body);
  }

  let res;
  try {
    res = await fetch(`${BASE}/api${path}`, { method, headers, body: payload });
  } catch {
    throw new ApiError('Cannot reach the server. Check your internet connection and try again.', 0);
  }

  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    throw new ApiError(data.message || `Something went wrong (error ${res.status}).`, res.status);
  }
  return data;
}

export const query = (params = {}) => {
  const search = new URLSearchParams();
  Object.entries(params).forEach(([k, v]) => {
    if (v !== '' && v !== null && v !== undefined) search.set(k, v);
  });
  const s = search.toString();
  return s ? `?${s}` : '';
};

// Turns stored image URLs into displayable URLs. Uploaded files are absolute Firebase URLs already.
export const imageSrc = (image) => image?.url || '';

  const [saving, setSaving] = useState(false);

  async function save(e) {

    }

    } catch (err) {
      toast.error(errorMessage(err));
    }
  }

  return (
    <div>
      <PageHeader title="My account" text="Your display name and login details." />
      <div className="grid gap-6 lg:grid-cols-2">
        <form onSubmit={save} className="card space-y-5">
          <h2 className="text-lg font-semibold">Profile</h2>

export default function Login() {
  const { login, resetPassword, me, loading } = useAuth();
  const toast = useToast();
  const navigate = useNavigate();
  const location = useLocation();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const [emailError, setEmailError] = useState('');
  useDocumentTitle('Admin login');

  if (!loading && me) return <Navigate to={location.state?.from || '/admin'} replace />;

  async function submit(e) {
    e.preventDefault();
    setError('');
    if (!isEmail(email.trim())) return setEmailError('Please enter a valid email address.');
    setEmailError('');
    if (!password) return setError('Please enter your password.');
    setBusy(true);
    try {
      await login(email, password);
      navigate(location.state?.from || '/admin', { replace: true });
    } catch (err) {
      setError(errorMessage(err));
    } finally {
      setBusy(false);
    }
  }

  async function forgot() {
    setEmailError('');
    if (!isEmail(email.trim())) return setEmailError('Enter your email above first, then click "Forgot password".');
    try {
      await resetPassword(email);
      toast.success('Password reset email sent. Check your inbox.');
    } catch (err) {
      toast.error(errorMessage(err));
    }
  }

  return (
    <div className="relative flex min-h-screen items-center justify-center overflow-hidden px-5 py-16">
      <div aria-hidden="true" className="pointer-events-none absolute left-1/2 top-0 h-[480px] w-[700px] -translate-x-1/2 rounded-full opacity-25 blur-3xl" style={{ background: 'radial-gradient(closest-side,var(--color-accent-2),transparent)' }} />

          <form onSubmit={submit} noValidate className="mt-8 space-y-5">
            <Field label="Email" htmlFor="login-email" error={emailError}>
              <Input id="login-email" type="email" autoComplete="username" value={email} onChange={(e) => setEmail(e.target.value)} invalid={!!emailError} />

import { useState } from 'react';

import PageHeader from './PageHeader.jsx';

export default function Messages() {
  const [unreadOnly, setUnreadOnly] = useState(false);
  const { data, loading, error, reload, setData } = useAsync(() => staffApi.messages(unreadOnly), [unreadOnly]);
  const [pendingDelete, setPendingDelete] = useState(null);
  const [busy, setBusy] = useState(false);
  const toast = useToast();
  useDocumentTitle('Messages');

  async function toggleRead(m) {
    try {
      const updated = await staffApi.setMessageRead(m.id, !m.read);
      setData((d) => ({
        ...d,
        items: d.items.map((x) => (x.id === m.id ? { ...x, read: updated.read } : x)),
        unread: d.unread + (updated.read ? -1 : 1),
      }));
    } catch (err) {
      toast.error(errorMessage(err));
    }
  }

  async function confirmDelete() {
    setBusy(true);
    try {
      await staffApi.deleteMessage(pendingDelete.id);
      setData((d) => ({
        ...d,
        items: d.items.filter((x) => x.id !== pendingDelete.id),
        unread: pendingDelete.read ? d.unread : Math.max(0, d.unread - 1),
      }));
      toast.success('Message deleted.');
      setPendingDelete(null);
    } catch (err) {
      toast.error(errorMessage(err));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div>
      <PageHeader
        title="Messages"
        text={data ? `${data.unread} unread Â· ${data.items.length} shown` : 'Messages sent through your contact form.'}
        actions={(
          <div className="flex gap-2" role="group" aria-label="Filter messages">
            <button className={`btn-sm ${!unreadOnly ? 'btn-primary' : 'btn-ghost'}`} aria-pressed={!unreadOnly} onClick={() => setUnreadOnly(false)}>All</button>
            <button className={`btn-sm ${unreadOnly ? 'btn-primary' : 'btn-ghost'}`} aria-pressed={unreadOnly} onClick={() => setUnreadOnly(true)}>Unread</button>
          </div>
        )}
      />

      {loading && <PageLoader />}
      {error && <ErrorState message={error} onRetry={reload} />}
      {data && data.items.length === 0 && <EmptyState title={unreadOnly ? 'No unread messages' : 'No messages yet'} text="Messages from your contact page will appear here." />}

      {data && data.items.length > 0 && (
        <ul className="space-y-4">
          {data.items.map((m) => (
            <li key={m.id} className={`card ${m.read ? '' : 'border-accent/40'}`}>
              <div className="flex flex-wrap items-start justify-between gap-4">
                <div className="min-w-0">
                  <div className="flex flex-wrap items-center gap-2">
                    <p className="font-semibold">{m.name}</p>
                    {!m.read && <span className="badge text-accent">Unread</span>}
                  </div>
                  <a href={`mailto:${m.email}`} className="text-sm text-muted hover:text-white">{m.email}</a>
                  <p className="mt-1 text-xs text-muted">{formatDateTime(m.createdAt)}</p>
                </div>
                <div className="flex flex-wrap gap-2">
                  <button className="btn-ghost btn-sm" onClick={() => toggleRead(m)}>{m.read ? 'Mark unread' : 'Mark read'}</button>
                  <a className="btn-ghost btn-sm" href={`mailto:${m.email}?subject=${encodeURIComponent(`Re: ${m.subject || 'Your message'}`)}`}>Reply</a>
                  <button className="btn-ghost btn-sm text-danger" onClick={() => setPendingDelete(m)}>Delete</button>
                </div>
              </div>

        busy={busy}

        onCancel={() => setPendingDelete(null)}

import { useMemo, useState } from 'react';
import { publicApi } from '../../lib/services.js';
import { useAsync } from '../../hooks/useAsync.js';
import { useDocumentTitle } from '../../hooks/useDocumentTitle.js';
import ProjectCard from '../../components/ProjectCard.jsx';
import Reveal from '../../components/Reveal.jsx';
import { EmptyState, ErrorState, SkeletonCards } from '../../components/States.jsx';

export default function Projects() {
  const { data, loading, error, reload } = useAsync(() => publicApi.projects(), []);
  const [category, setCategory] = useState('All');
  const [search, setSearch] = useState('');
  useDocumentTitle('Work');

  const items = data?.items || [];
  const categories = useMemo(() => ['All', ...new Set(items.map((p) => p.category).filter(Boolean))], [items]);

  const shown = useMemo(() => {
    const q = search.trim().toLowerCase();
    return items.filter((p) => {
      if (category !== 'All' && p.category !== category) return false;
      if (!q) return true;
      return [p.title, p.description, p.client, p.category, ...(p.tags || [])].join(' ').toLowerCase().includes(q);
    });
  }, [items, category, search]);

  return (
    <div className="pt-28 sm:pt-32">
      <section className="container-x">
        <Reveal>
          <p className="eyebrow">Portfolio</p>
          <h1 className="mt-3 text-4xl font-extrabold sm:text-5xl">Selected work</h1>
          <p className="mt-4 max-w-2xl text-muted">A collection of posters, social media creatives and visual identity projects.</p>
        </Reveal>

        <div className="mt-10 flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
          <div className="flex flex-wrap gap-2" role="group" aria-label="Filter by category">
            {categories.map((c) => (
              <button
                key={c}
                type="button"
                aria-pressed={category === c}
                onClick={() => setCategory(c)}
                className={`rounded-full px-4 py-2 text-sm font-medium transition ${category === c ? 'bg-accent text-white' : 'border border-line text-muted hover:text-white'}`}
              >
                {c}
              </button>
            ))}
          </div>
          <input
            type="search"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="Search projects..."
            aria-label="Search projects"
            className="input md:max-w-xs"
          />
        </div>

        <div className="mt-12 pb-12">
          {loading && <SkeletonCards count={6} />}
          {error && <ErrorState message={error} onRetry={reload} />}
          {!loading && !error && items.length === 0 && (
            <EmptyState title="No projects yet" text="New work will appear here soon." />
          )}
          {!loading && !error && items.length > 0 && shown.length === 0 && (
            <EmptyState title="No matching projects" text="Try a different search or category." />
          )}
          {shown.length > 0 && (
            <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
              {shown.map((p, i) => (
                <Reveal key={p.id} delay={(i % 3) * 80}><ProjectCard project={p} /></Reveal>
              ))}
            </div>
          )}
        </div>
      </section>
    </div>
  );
}
